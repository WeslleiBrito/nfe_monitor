import sys

from PySide6.QtCore import QCoreApplication, QUrl
from PySide6.QtGui import QGuiApplication
from PySide6.QtQml import QQmlApplicationEngine
from PySide6.QtQuickControls2 import QQuickStyle

from nfe_monitor.infrastructure.paths import recurso
from nfe_monitor.ui.themes.theme_manager import ThemeManager
from nfe_monitor.ui.viewmodels.config_viewmodel import ConfigViewModel
from nfe_monitor.ui.viewmodels.main_viewmodel import MainViewModel


def main() -> int:
    QCoreApplication.setOrganizationName("NFeMonitor")
    QCoreApplication.setApplicationName("NF-e Monitor")
    QQuickStyle.setStyle("Basic")  # personalização total dos controles

    app = QGuiApplication(sys.argv)
    engine = QQmlApplicationEngine()

    tema = ThemeManager(app)
    view_model = MainViewModel()
    config = ConfigViewModel()
    config.salvo.connect(view_model.atualizarCertificado)

    ctx = engine.rootContext()
    ctx.setContextProperty("Theme", tema)
    ctx.setContextProperty("viewModel", view_model)
    ctx.setContextProperty("config", config)

    engine.load(QUrl.fromLocalFile(str(recurso("nfe_monitor/ui/qml/Main.qml"))))
    if not engine.rootObjects():
        return 1

    codigo = app.exec()
    del engine  # destrói o QML antes dos view models
    return codigo


if __name__ == "__main__":
    sys.exit(main())

import logging
import shutil
from datetime import datetime
from pathlib import Path

from nfe_monitor.domain.resumo import FUSO
from nfe_monitor.infrastructure.database import notas_repository as repo
from nfe_monitor.infrastructure.database.conexao import conectar
from nfe_monitor.infrastructure.paths import CNPJ, PASTA_XML
from nfe_monitor.infrastructure.sefaz import cliente, distribuicao
from nfe_monitor.infrastructure.sefaz.manifestacao import (
    CSTAT_DUPLICIDADE, CSTATS_CIENCIA_OK, ErroEvento, enviar_ciencia,
)
from nfe_monitor.infrastructure.sefaz.parametros import Config

__all__ = [
    "ErroEvento", "manifestar_ciencia", "baixar_xml_completo", "obter_xml",
    "caminho_livre", "baixar_e_copiar",
]

log = logging.getLogger("nfe_monitor")


def manifestar_ciencia(con, cnpj: str, chave: str, cfg: Config) -> None:
    inicio = datetime.now(FUSO)
    try:
        cstat, motivo = enviar_ciencia(chave, cnpj, cfg)
    except Exception as e:
        with con:
            repo.registrar_operacao(con, "ciencia", chave, inicio, "erro", str(e))
        raise

    if cstat in CSTATS_CIENCIA_OK or cstat == CSTAT_DUPLICIDADE:
        with con:
            repo.atualizar_manifestacao(con, chave, "ciencia")
            repo.registrar_operacao(con, "ciencia", chave, inicio, "ok", f"{cstat} {motivo}")
        log.info("Ciência registrada para %s: %s %s", chave, cstat, motivo)
        return

    with con:
        repo.registrar_operacao(con, "ciencia", chave, inicio, "erro", f"{cstat} {motivo}")
    raise ErroEvento(f"A SEFAZ recusou a ciência: {cstat} - {motivo}")


def baixar_xml_completo(con, cnpj: str, chave: str, cfg: Config) -> Path:
    inicio = datetime.now(FUSO)
    try:
        corpo = cliente.consultar_chave(cnpj, chave, cfg)
        lote = distribuicao.ler_lote(corpo, exigir_nsu=False)

        if lote.cstat == distribuicao.CSTAT_ESPERA:
            raise ErroEvento(f"SEFAZ pediu espera (656): {lote.motivo}")
        if lote.cstat not in (distribuicao.CSTAT_SEM_DOCS, distribuicao.CSTAT_COM_DOCS):
            raise ErroEvento(f"SEFAZ respondeu cStat {lote.cstat}: {lote.motivo}")

        for doc in lote.docs:
            if doc.tipo != "completo":
                continue
            resumo = distribuicao.ler_completo(doc.xml)
            if resumo.chave == chave:
                with con:
                    repo.salvar_completo(con, resumo, doc.xml, PASTA_XML)
                    repo.registrar_operacao(con, "consulta_chave", chave, inicio, "ok",
                                            f"{lote.cstat} {lote.motivo}")
                return PASTA_XML / f"{chave}.xml"

        raise ErroEvento(
            "A SEFAZ não retornou o XML completo desta nota. "
            f"Resposta: {lote.cstat} {lote.motivo}"
        )
    except Exception as e:
        with con:
            repo.registrar_operacao(con, "consulta_chave", chave, inicio, "erro", str(e))
        raise


def obter_xml(con, cnpj: str, chave: str, cfg: Config | None = None) -> Path:
    """Fluxo completo para uma nota resumo: ciência (se preciso) e depois o XML completo."""
    cfg = cfg or Config()
    if not repo.tem_ciencia(con, chave):
        manifestar_ciencia(con, cnpj, chave, cfg)
    return baixar_xml_completo(con, cnpj, chave, cfg)


def caminho_livre(pasta: Path, nome: str) -> Path:
    destino = pasta / nome
    base, ext, n = destino.stem, destino.suffix, 1
    while destino.exists():
        destino = pasta / f"{base} ({n}){ext}"
        n += 1
    return destino


def baixar_e_copiar(chave: str, pasta: Path) -> Path:
    cfg = Config()
    con = conectar(cfg.banco)
    try:
        origem = obter_xml(con, CNPJ, chave, cfg)
    finally:
        con.close()
    destino = caminho_livre(pasta, f"{chave}.xml")
    shutil.copy2(origem, destino)
    return destino

import logging
import sys
from dataclasses import dataclass, field

from nfe_monitor.application.sincronizar import cnpj_valido
from nfe_monitor.infrastructure import brasilapi, certificados, preferencias
from nfe_monitor.infrastructure.database import notas_repository as repo
from nfe_monitor.infrastructure.database.conexao import UF_CODIGO, conectar
from nfe_monitor.infrastructure.sefaz.parametros import Config

__all__ = ["DIAS_AVISO", "ErroCadastro", "Cadastro", "cadastrar_por_certificado"]

log = logging.getLogger("nfe_monitor")

DIAS_AVISO = 30


class ErroCadastro(ValueError):
    """Certificado inadequado para cadastrar a empresa (mostrado ao usuário)."""


@dataclass
class Cadastro:
    empresa: repo.Empresa
    nova: bool
    avisos: list[str] = field(default_factory=list)


def cadastrar_por_certificado(thumbprint: str, cfg: Config | None = None,
                              uf: str | None = None, definir_padrao: bool = True) -> Cadastro:
    """Cadastra (ou atualiza) a empresa do certificado. `uf` sobrepõe a BrasilAPI."""
    thumbprint = thumbprint.strip().replace(" ", "").upper()
    if not thumbprint:
        raise ErroCadastro("Nenhum thumbprint informado nem salvo no config.json")
    cert = next((c for c in certificados.listar() if c.thumbprint == thumbprint), None)
    if cert is None:
        raise ErroCadastro(f"Certificado {thumbprint} não encontrado no Windows")
    if not cert.cnpj or not cnpj_valido(cert.cnpj):
        raise ErroCadastro("O certificado não é um e-CNPJ válido")
    if cert.vencido:
        raise ErroCadastro(f"Certificado vencido em {cert.validade:%d/%m/%Y}")

    avisos = []
    if not cert.tem_chave:
        avisos.append("Windows não informou chave privada (token A3 desconectado?)")
    if cert.dias_restantes <= DIAS_AVISO:
        avisos.append(f"O certificado vence em {cert.dias_restantes} dia(s) "
                      f"({cert.validade:%d/%m/%Y})")

    if uf is not None:
        uf = uf.strip().upper()
        if uf not in UF_CODIGO:
            raise ErroCadastro(f"UF inválida: {uf}")

    cfg = cfg or Config()
    con = conectar(cfg.banco)
    try:
        existente = repo.obter_empresa(con, cert.cnpj)
        api = {} if uf else (brasilapi.consultar_cnpj(cert.cnpj) or {})
        if not uf and not api:
            avisos.append("BrasilAPI indisponível: dados tirados do certificado")

        uf_api = api.get("uf") if api.get("uf") in UF_CODIGO else None
        uf_final = uf or uf_api or (existente.uf if existente else None)
        if not uf_final:
            avisos.append("UF não identificada: informe com --uf para liberar a consulta")

        empresa = repo.Empresa(
            cnpj=cert.cnpj,
            razao_social=api.get("razao_social") or cert.nome
                         or (existente.razao_social if existente else None),
            uf=uf_final,
            thumbprint=cert.thumbprint,
            origem="windows",
            ativo=True,
        )
        with con:
            repo.salvar_empresa(con, empresa)
    finally:
        con.close()

    if definir_padrao:
        p = preferencias.carregar()
        p.thumbprint = cert.thumbprint
        preferencias.salvar(p)

    return Cadastro(empresa, existente is None, avisos)


def _ler_args(args: list[str]) -> tuple[str | None, str | None]:
    """Separa (thumbprint, uf). Thumbprint ausente ou 'atual' = o do config.json."""
    uf, posicionais, i = None, [], 0
    while i < len(args):
        if args[i] == "--uf" and i + 1 < len(args):
            uf, i = args[i + 1], i + 2
            continue
        posicionais.append(args[i])
        i += 1
    thumb = posicionais[0] if posicionais else None
    if thumb is None or thumb.lower() == "atual":
        thumb = preferencias.carregar().thumbprint or None
    return thumb, uf


def _listar_certificados() -> int:
    lista = certificados.listar()
    atual = preferencias.carregar().thumbprint.upper()
    if not lista:
        print("Nenhum certificado encontrado em CurrentUser\\My nem LocalMachine\\My.")
        print("  - Certificado A3: conecte o token/cartão e rode de novo.")
        print("  - Certificado A1: importe o .pfx (duplo clique > Usuário atual).")
        if atual:
            print(f"  Thumbprint salvo no config.json: {atual}")
        return 1
    for i, c in enumerate(lista, 1):
        marcas = []
        if c.thumbprint == atual:
            marcas.append("ATUAL")
        if c.vencido:
            marcas.append("VENCIDO")
        elif c.dias_restantes <= DIAS_AVISO:
            marcas.append("VENCE LOGO")
        if not c.tem_chave:
            marcas.append("SEM CHAVE")
        extra = f" [{', '.join(marcas)}]" if marcas else ""
        print(f"{i:>2}. {c.thumbprint}  {c.descricao}{extra}")
    return 0


def _main(argv: list[str]) -> int:
    uso = ("Uso:\n"
           "  python -m nfe_monitor.application.empresas certificados\n"
           "  python -m nfe_monitor.application.empresas cadastrar [THUMBPRINT|atual] [--uf BA]\n"
           "  python -m nfe_monitor.application.empresas listar")
    if not argv:
        print(uso)
        return 1

    comando, args = argv[0], argv[1:]
    try:
        if comando == "certificados":
            return _listar_certificados()

        if comando == "cadastrar":
            thumb, uf = _ler_args(args)
            res = cadastrar_por_certificado(thumb or "", uf=uf)
            e = res.empresa
            print(f"{'Cadastrada' if res.nova else 'Atualizada'}: "
                  f"{e.cnpj} {e.razao_social or '(sem nome)'} ({e.uf or 'sem UF'})")
            for a in res.avisos:
                print(f"  Aviso: {a}")
            return 0

        if comando == "listar":
            con = conectar(Config().banco)
            try:
                for e in repo.listar_empresas(con, somente_ativas=False):
                    estado = "ativa" if e.ativo else "inativa"
                    pronto = "ok" if e.completa else "incompleta"
                    print(f"{e.cnpj}  {e.uf or '--'}  {estado:<7} {pronto:<10} "
                          f"{e.razao_social or '(sem nome)'}")
            finally:
                con.close()
            return 0
    except (ErroCadastro, certificados.ErroCertificados) as e:
        log.error("%s", e)
        return 1

    print(uso)
    return 1


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
    sys.exit(_main(sys.argv[1:]))

from datetime import datetime, timedelta

from nfe_monitor.domain.resumo import FUSO
from nfe_monitor.infrastructure import preferencias

INTERVALO = timedelta(hours=1)


def ultima_sincronizacao() -> datetime | None:
    texto = preferencias.carregar().ultima_sincronizacao
    return datetime.fromisoformat(texto) if texto else None


def segundos_restantes() -> int:
    ultima = ultima_sincronizacao()
    if ultima is None:
        return 0
    restante = (ultima + INTERVALO) - datetime.now(FUSO)
    return max(0, int(restante.total_seconds()))


def registrar_disparo() -> None:
    """Grava o momento do clique. Vale também para sincronizações que falharam,
    porque a SEFAZ conta cada consulta, mesmo as que dão erro."""
    p = preferencias.carregar()
    p.ultima_sincronizacao = datetime.now(FUSO).isoformat(timespec="seconds")
    preferencias.salvar(p)

def consultar_notas(fornecedor: str, inicio: str | None, fim: str | None,
                    pagina: int, empresa_cnpj: str | None = None) -> tuple[list[dict], dict]:
    """Consulta somente a página pedida e calcula as métricas dos filtros ativos.

    empresa_cnpj vazio ou None = notas de todas as empresas.
    """

        return repo.buscar_pagina(con, fornecedor, inicio, fim,
                                  ITENS_POR_PAGINA, pagina * ITENS_POR_PAGINA,
                                  empresa_cnpj or None)
    finally:
        con.close()

import logging
import sys
from dataclasses import dataclass, replace
from datetime import datetime

from nfe_monitor.domain.resumo import FUSO
from nfe_monitor.infrastructure.database import notas_repository as repo
from nfe_monitor.infrastructure.database.conexao import UF_CODIGO, conectar
from nfe_monitor.infrastructure.paths import PASTA_XML
from nfe_monitor.infrastructure.sefaz import cliente, distribuicao
from nfe_monitor.infrastructure.sefaz.parametros import Config

__all__ = [
    "CSTATS_OK", "MINUTOS_FIM_FILA", "MINUTOS_656",
    "EsperaSefaz", "CnpjInvalido", "Resultado",
    "cnpj_valido", "sincronizar", "sincronizar_todas",
]

log = logging.getLogger("nfe_monitor")

CSTATS_OK = (distribuicao.CSTAT_SEM_DOCS, distribuicao.CSTAT_COM_DOCS)
MINUTOS_FIM_FILA = 60  # regra da SEFAZ: aguardar 1h ao zerar a fila
MINUTOS_656 = 60


class EsperaSefaz(RuntimeError):
    """A SEFAZ pediu para aguardar (656) ou a empresa ainda está no intervalo de espera."""


class CnpjInvalido(ValueError):
    """CNPJ com dígito verificador errado ou não cadastrado."""


@dataclass
class Resultado:
    cnpj: str
    novas: int = 0
    status: str = "ok"          # ok, espera ou erro
    detalhe: str = ""


def cnpj_valido(cnpj: str) -> bool:
    """Confere formato e dígitos verificadores do CNPJ."""
    if len(cnpj) != 14 or not cnpj.isdigit() or len(set(cnpj)) == 1:
        return False

    def dv(base: str) -> str:
        pesos = list(range(len(base) - 7, 1, -1)) + list(range(9, 1, -1))
        resto = sum(int(d) * p for d, p in zip(base, pesos)) % 11
        return "0" if resto < 2 else str(11 - resto)

    d1 = dv(cnpj[:12])
    return cnpj[12:] == d1 + dv(cnpj[:12] + d1)


def _config_da_empresa(e: repo.Empresa, cfg: Config) -> Config:
    """Usa o certificado e a UF da empresa; o que faltar vem do Config atual."""
    return replace(
        cfg,
        thumbprint=e.thumbprint or cfg.thumbprint,
        uf_autor=UF_CODIGO[e.uf] if e.uf else cfg.uf_autor,
    )


def _processar_lote(con, cnpj: str, lote: distribuicao.Lote) -> int:
    novas = 0
    for doc in lote.docs:
        if doc.tipo == "resumo":
            if repo.inserir_resumo(con, distribuicao.ler_resumo(doc.xml), cnpj):
                novas += 1
        elif doc.tipo == "completo":
            r = distribuicao.ler_completo(doc.xml)
            if repo.salvar_completo(con, r, doc.xml, PASTA_XML, cnpj):
                novas += 1
    return novas


def sincronizar(cnpj: str, cfg: Config | None = None, forcar: bool = False) -> int:
    """Consulta a distribuição a partir do último NSU. Retorna quantas notas novas entraram."""
    cnpj = "".join(c for c in cnpj if c.isdigit())
    if not cnpj_valido(cnpj):
        raise CnpjInvalido(f"CNPJ inválido: {cnpj or '(vazio)'}")

    cfg = cfg or Config()
    con = conectar(cfg.banco)
    inicio = datetime.now(FUSO)
    novas = 0
    try:
        empresa = repo.obter_empresa(con, cnpj)
        if empresa is None:
            raise CnpjInvalido(f"CNPJ {cnpj} não está cadastrado")
        if not forcar and (ate := repo.bloqueada_ate(con, cnpj)):
            raise EsperaSefaz(f"Aguardando a SEFAZ até {ate:%d/%m %H:%M}")

        cfg_emp = _config_da_empresa(empresa, cfg)
        ult = repo.ler_ult_nsu(con, cnpj)
        log.info("Consultando %s a partir do NSU %s", cnpj, ult)

        while True:
            lote = distribuicao.ler_lote(cliente.consultar(cnpj, ult, cfg_emp))

            if lote.cstat == distribuicao.CSTAT_ESPERA:
                with con:
                    ate = repo.bloquear_empresa(con, cnpj, MINUTOS_656)
                    repo.registrar_operacao(con, "sincronizacao", None, inicio, "erro",
                                            f"{cnpj}: 656 {lote.motivo}")
                raise EsperaSefaz(f"{lote.motivo} (liberado às {ate:%H:%M})")
            if lote.cstat not in CSTATS_OK:
                raise RuntimeError(f"SEFAZ respondeu cStat {lote.cstat}: {lote.motivo}")

            # Lote e NSU na mesma transação: se falhar no meio, nada é perdido.
            with con:
                novas += _processar_lote(con, cnpj, lote)
                repo.salvar_ult_nsu(con, cnpj, lote.ult_nsu, lote.max_nsu)

            log.info("Lote %s: cStat=%s, %d docs, NSU %s/%s",
                     cnpj, lote.cstat, len(lote.docs), lote.ult_nsu, lote.max_nsu)

            if lote.cstat == distribuicao.CSTAT_SEM_DOCS or lote.ult_nsu == lote.max_nsu:
                with con:
                    repo.bloquear_empresa(con, cnpj, MINUTOS_FIM_FILA)
                break
            if lote.ult_nsu == ult:  # NSU não avançou: evita laço infinito
                break
            ult = lote.ult_nsu

        with con:
            repo.registrar_operacao(con, "sincronizacao", None, inicio, "ok",
                                    f"{cnpj}: {novas} novas")
    except (EsperaSefaz, CnpjInvalido):
        raise
    except Exception as e:
        with con:
            repo.registrar_operacao(con, "sincronizacao", None, inicio, "erro",
                                    f"{cnpj}: {e}"[:500])
        raise
    finally:
        con.close()

    return novas


def sincronizar_todas(cfg: Config | None = None) -> list[Resultado]:
    """Sincroniza todas as empresas ativas. Uma falha não interrompe as demais."""
    cfg = cfg or Config()
    con = conectar(cfg.banco)
    try:
        cnpjs = [e.cnpj for e in repo.listar_empresas(con)]
    finally:
        con.close()

    resultados = []
    for cnpj in cnpjs:
        res = Resultado(cnpj)
        try:
            res.novas = sincronizar(cnpj, cfg)
        except EsperaSefaz as e:
            res.status, res.detalhe = "espera", str(e)
        except Exception as e:
            res.status, res.detalhe = "erro", str(e)
            log.exception("Falha ao sincronizar %s", cnpj)
        resultados.append(res)
    return resultados


def _main(argv: list[str]) -> int:
    forcar = "--forcar" in argv
    args = [a for a in argv if a != "--forcar"]

    if len(args) > 1:
        print("Uso: python -m nfe_monitor.application.sincronizar [CNPJ] [--forcar]")
        return 1

    if args:
        try:
            print(f"Notas novas: {sincronizar(args[0], forcar=forcar)}")
            return 0
        except CnpjInvalido as e:
            log.error("%s", e)
            return 1
        except EsperaSefaz as e:
            log.warning("SEFAZ pediu espera: %s", e)
            return 2
        except Exception as e:
            log.error("Falha na sincronização: %s", e)
            return 3

    resultados = sincronizar_todas()
    if not resultados:
        print("Nenhuma empresa cadastrada.")
        return 1
    for r in resultados:
        print(f"{r.cnpj}  {r.status:<7} novas={r.novas}  {r.detalhe}")
    return 0 if all(r.status != "erro" for r in resultados) else 3


if __name__ == "__main__":

from urllib.request import Request, urlopen

URL = "https://brasilapi.com.br/api/cnpj/v1/{cnpj}"


def consultar_cnpj(cnpj: str, timeout: int = 10) -> dict | None:
    """Devolve {'razao_social', 'uf'} ou None se a API falhar (o cadastro segue sem ela)."""
    req = Request(URL.format(cnpj=cnpj), headers={"User-Agent": "nfe-monitor"})
    try:
        with urlopen(req, timeout=timeout) as resp:
            dados = json.loads(resp.read().decode("utf-8"))

        return None
    return {
        "razao_social": (dados.get("razao_social") or "").strip() or None,
        "uf": (dados.get("uf") or "").strip().upper() or None,
    }

import base64
import json
import os
import re
import subprocess
from dataclasses import dataclass
from datetime import datetime

__all__ = ["AVISO_DIAS", "Certificado", "ErroCertificados", "extrair_cnpj",
           "listar", "selecionado", "validar_para_uso"]

AVISO_DIAS = 30

_SEM_JANELA = getattr(subprocess, "CREATE_NO_WINDOW", 0)
_CN_ECNPJ = re.compile(r"CN=([^,]*?):(\d{14})")
_CN = re.compile(r"CN=([^,]+)")
_QUATORZE = re.compile(r"(?<!\d)(\d{14})(?!\d)")

# Usa X509Store direto: não depende da unidade Cert: (que falha no 5.1 aberto pelo pwsh 7).
_SCRIPT = r"""
$ErrorActionPreference = 'Stop'
[Console]::OutputEncoding = [Text.Encoding]::UTF8
$lista = New-Object System.Collections.ArrayList
foreach ($loja in 'CurrentUser', 'LocalMachine') {
    $store = New-Object System.Security.Cryptography.X509Certificates.X509Store('My', $loja)
    try {
        $store.Open('ReadOnly, OpenExistingOnly')
    } catch {
        continue
    }
    foreach ($c in $store.Certificates) {
        [void]$lista.Add([pscustomobject]@{
            thumbprint = $c.Thumbprint
            subject    = $c.Subject
            validade   = $c.NotAfter.ToString('yyyy-MM-dd HH:mm:ss')
            loja       = $loja
            chave      = [bool]$c.HasPrivateKey
        })
    }
    $store.Close()
}
ConvertTo-Json -InputObject @($lista) -Compress
"""


class ErroCertificados(RuntimeError):
    """Falha ao ler o repositório de certificados do Windows."""


@dataclass(frozen=True)
class Certificado:
    thumbprint: str
    subject: str
    validade: datetime
    cnpj: str | None
    nome: str | None
    loja: str = "CurrentUser"
    tem_chave: bool = True

    @property
    def dias_restantes(self) -> int:
        return (self.validade - datetime.now()).days

    @property
    def vencido(self) -> bool:
        return self.validade <= datetime.now()

    @property
    def titular(self) -> str:
        if self.nome:
            return self.nome
        m = _CN.search(self.subject)
        return m.group(1).strip() if m else self.subject

    @property
    def situacao(self) -> str:
        return "Vencido" if self.vencido else "Válido"

    @property
    def descricao(self) -> str:
        return (f"{self.titular} ({self.cnpj or 'sem CNPJ'}) - vence {self.validade:%d/%m/%Y}"
                f" [{self.loja}]")


def extrair_cnpj(subject: str) -> tuple[str | None, str | None]:
    """Devolve (cnpj, nome) do subject. e-CNPJ usa 'CN=RAZAO SOCIAL:CNPJ'."""
    m = _CN_ECNPJ.search(subject)
    if m:
        return m.group(2), m.group(1).strip() or None
    m = _QUATORZE.search(subject)
    return (m.group(1) if m else None), None


def _ambiente_limpo() -> dict:
    """Remove o PSModulePath herdado do PowerShell 7, que quebra o Windows PowerShell 5.1."""
    env = dict(os.environ)
    env.pop("PSModulePath", None)
    return env


def listar() -> list[Certificado]:
    codificado = base64.b64encode(_SCRIPT.encode("utf-16-le")).decode("ascii")
    try:
        r = subprocess.run(
            ["powershell", "-NoProfile", "-NonInteractive", "-EncodedCommand", codificado],
            capture_output=True, timeout=30, creationflags=_SEM_JANELA, env=_ambiente_limpo(),
        )
    except subprocess.TimeoutExpired as e:
        raise ErroCertificados("Tempo esgotado ao listar certificados") from e

    erro = r.stderr.decode("utf-8", errors="replace").strip()
    if r.returncode != 0:
        raise ErroCertificados(erro or f"PowerShell saiu com código {r.returncode}")

    texto = r.stdout.decode("utf-8-sig", errors="replace").strip() or "[]"
    try:
        dados = json.loads(texto)
    except json.JSONDecodeError as e:
        raise ErroCertificados(f"Saída inesperada do PowerShell: {texto[:200]!r} {erro[:200]}") from e
    if isinstance(dados, dict):
        dados = [dados]

    # Mesmo certificado nas duas lojas: fica o que tem chave privada.
    unicos: dict[str, Certificado] = {}
    for d in dados:
        cnpj, nome = extrair_cnpj(d["subject"])
        c = Certificado(
            thumbprint=d["thumbprint"].upper(),
            subject=d["subject"],
            validade=datetime.strptime(d["validade"], "%Y-%m-%d %H:%M:%S"),
            cnpj=cnpj,
            nome=nome,
            loja=d.get("loja", "CurrentUser"),
            tem_chave=bool(d.get("chave", True)),
        )
        atual = unicos.get(c.thumbprint)
        if atual is None or (c.tem_chave and not atual.tem_chave):
            unicos[c.thumbprint] = c
    return sorted(unicos.values(), key=lambda c: c.validade, reverse=True)


def selecionado() -> Certificado | None:
    """Certificado salvo no config.json, se ainda estiver instalado."""
    from nfe_monitor.infrastructure import preferencias
    thumb = preferencias.carregar().thumbprint.strip().upper()
    if not thumb:
        return None
    return next((c for c in listar() if c.thumbprint == thumb), None)


def validar_para_uso() -> Certificado:
    cert = selecionado()
    if cert is None:
        raise ErroCertificados("Nenhum certificado configurado ou ele não está mais instalado. "
                               "Abra as Configurações e selecione um.")
    if cert.vencido:
        raise ErroCertificados(f"O certificado venceu em {cert.validade:%d/%m/%Y}.")
    return cert

import sqlite3
from pathlib import Path

from nfe_monitor.infrastructure.sefaz.distribuicao import normalizar_data

__all__ = ["SCHEMA", "UF_CODIGO", "conectar"]

UF_CODIGO = {
    "RO": "11", "AC": "12", "AM": "13", "RR": "14", "PA": "15", "AP": "16",
    "TO": "17", "MA": "21", "PI": "22", "CE": "23", "RN": "24", "PB": "25",
    "PE": "26", "AL": "27", "SE": "28", "BA": "29", "MG": "31", "ES": "32",
    "RJ": "33", "SP": "35", "PR": "41", "SC": "42", "RS": "43", "MS": "50",
    "MT": "51", "GO": "52", "DF": "53",
}

SCHEMA = """
CREATE TABLE IF NOT EXISTS empresas (
    cnpj          TEXT PRIMARY KEY,
    razao_social  TEXT,
    uf            TEXT,              -- sigla; NULL até o cadastro ser completado
    thumbprint    TEXT,              -- certificado no repositório do Windows
    origem        TEXT NOT NULL DEFAULT 'windows',   -- 'windows' ou caminho do .pfx
    ativo         INTEGER NOT NULL DEFAULT 1,
    criado_em     TEXT NOT NULL DEFAULT (datetime('now', 'localtime'))
);
CREATE TABLE IF NOT EXISTS nfe (
    chave              TEXT PRIMARY KEY,
    empresa_cnpj       TEXT,            -- empresa monitorada que recebeu a nota
    emitente_cnpj      TEXT NOT NULL,
    emitente_nome      TEXT,
    data_emissao       TEXT,            -- 'AAAA-MM-DD HH:MM:SS', horário de Brasília
    valor              TEXT,            -- guardado como texto para preservar casas decimais
    situacao           TEXT,            -- cSitNFe no resumo; cStat do protocolo no completo
    protocolo          TEXT,
    manifestacao       TEXT NOT NULL DEFAULT 'pendente',
    caminho_xml        TEXT,            -- preenchido só quando o XML completo foi baixado
    tipo               TEXT NOT NULL DEFAULT 'resumo',  -- 'resumo' ou 'completo'
    criado_em          TEXT NOT NULL DEFAULT (datetime('now', 'localtime'))
);
CREATE TABLE IF NOT EXISTS controle_nsu (
    cnpj           TEXT PRIMARY KEY,
    ult_nsu        TEXT NOT NULL,
    max_nsu        TEXT,
    bloqueado_ate  TEXT,             -- 'AAAA-MM-DD HH:MM:SS'; preenchido após cStat 656
    atualizado_em  TEXT
);
CREATE TABLE IF NOT EXISTS operacoes (
    id       INTEGER PRIMARY KEY AUTOINCREMENT,
    tipo     TEXT NOT NULL,      -- sincronizacao, ciencia, consulta_chave, download_xml
    chave    TEXT,               -- NULL em sincronizacao
    inicio   TEXT NOT NULL,      -- 'AAAA-MM-DD HH:MM:SS', horário de Brasília
    fim      TEXT NOT NULL,
    status   TEXT NOT NULL,      -- ok ou erro
    detalhe  TEXT                -- cStat e motivo, ou mensagem de erro
);
"""

INDICES = """
CREATE INDEX IF NOT EXISTS ix_nfe_empresa ON nfe(empresa_cnpj);
"""


def conectar(caminho: Path) -> sqlite3.Connection:
    con = sqlite3.connect(caminho)
    con.executescript(SCHEMA)
    _migrar(con)
    con.executescript(INDICES)
    return con


def _colunas(con: sqlite3.Connection, tabela: str) -> set[str]:
    return {linha[1] for linha in con.execute(f"PRAGMA table_info({tabela})")}


def _adicionar_coluna(con: sqlite3.Connection, tabela: str, definicao: str) -> None:
    nome = definicao.split()[0]
    if nome not in _colunas(con, tabela):
        con.execute(f"ALTER TABLE {tabela} ADD COLUMN {definicao}")


def _migrar(con: sqlite3.Connection) -> None:
    """Atualiza bancos criados por versões anteriores. Seguro de executar sempre."""
    _adicionar_coluna(con, "nfe", "tipo TEXT NOT NULL DEFAULT 'resumo'")
    _adicionar_coluna(con, "nfe", "empresa_cnpj TEXT")
    _adicionar_coluna(con, "controle_nsu", "max_nsu TEXT")
    _adicionar_coluna(con, "controle_nsu", "bloqueado_ate TEXT")
    _adicionar_coluna(con, "controle_nsu", "atualizado_em TEXT")

    # Datas gravadas antes da normalização ainda têm 'T' e fuso.
    antigas = con.execute(
        "SELECT chave, data_emissao FROM nfe WHERE data_emissao LIKE '%T%'"
    ).fetchall()
    for chave, data in antigas:
        con.execute(
            "UPDATE nfe SET data_emissao = ? WHERE chave = ?",
            (normalizar_data(data), chave),
        )

    # Todo CNPJ com NSU vira empresa (cadastro incompleto até a Etapa 2).
    con.execute(
        "INSERT OR IGNORE INTO empresas (cnpj) SELECT cnpj FROM controle_nsu"
    )

    # Banco de versão anterior com uma única empresa: as notas antigas são dela.
    cnpjs = [linha[0] for linha in con.execute("SELECT cnpj FROM empresas")]
    if len(cnpjs) == 1:
        con.execute(
            "UPDATE nfe SET empresa_cnpj = ? WHERE empresa_cnpj IS NULL", (cnpjs[0],)
        )
    con.commit()

NSU_ZERO = "0" * 15
_FMT = "%Y-%m-%d %H:%M:%S"


def _agora() -> str:
    return datetime.now(FUSO).strftime(_FMT)


def _cursor(con: sqlite3.Connection) -> sqlite3.Cursor:
    """Cursor com acesso por nome, sem alterar o row_factory da conexão."""
    cur = con.cursor()
    cur.row_factory = sqlite3.Row
    return cur


# ---------------------------------------------------------------- empresas
@dataclass
class Empresa:
    cnpj: str
    razao_social: str | None = None
    uf: str | None = None
    thumbprint: str | None = None
    origem: str = "windows"
    ativo: bool = True

    @property
    def completa(self) -> bool:
        """Cadastro pronto para consultar a SEFAZ."""
        return bool(self.uf and self.thumbprint)

    @property
    def c_uf_autor(self) -> str:
        if not self.uf:
            raise ValueError(f"Empresa {self.cnpj} sem UF cadastrada")
        return UF_CODIGO[self.uf]


def _para_empresa(r: sqlite3.Row) -> Empresa:
    return Empresa(r["cnpj"], r["razao_social"], r["uf"], r["thumbprint"],
                   r["origem"], bool(r["ativo"]))


def salvar_empresa(con: sqlite3.Connection, e: Empresa) -> None:
    if len(e.cnpj) != 14 or not e.cnpj.isdigit():
        raise ValueError(f"CNPJ inválido: {e.cnpj}")
    if e.uf is not None and e.uf not in UF_CODIGO:
        raise ValueError(f"UF inválida: {e.uf}")
    con.execute(
        """INSERT INTO empresas (cnpj, razao_social, uf, thumbprint, origem, ativo)
           VALUES (?, ?, ?, ?, ?, ?)
           ON CONFLICT(cnpj) DO UPDATE SET
               razao_social = excluded.razao_social, uf = excluded.uf,
               thumbprint = excluded.thumbprint, origem = excluded.origem,
               ativo = excluded.ativo""",
        (e.cnpj, e.razao_social, e.uf, e.thumbprint, e.origem, int(e.ativo)),
    )
    con.execute(
        "INSERT OR IGNORE INTO controle_nsu (cnpj, ult_nsu, atualizado_em) VALUES (?, ?, ?)",
        (e.cnpj, NSU_ZERO, _agora()),
    )


def obter_empresa(con: sqlite3.Connection, cnpj: str) -> Empresa | None:
    r = _cursor(con).execute("SELECT * FROM empresas WHERE cnpj = ?", (cnpj,)).fetchone()
    return _para_empresa(r) if r else None


def listar_empresas(con: sqlite3.Connection, somente_ativas: bool = True) -> list[Empresa]:
    sql = "SELECT * FROM empresas" + (" WHERE ativo = 1" if somente_ativas else "")
    return [_para_empresa(r) for r in _cursor(con).execute(sql + " ORDER BY razao_social")]


def desativar_empresa(con: sqlite3.Connection, cnpj: str) -> None:
    """Para de monitorar sem apagar as notas já baixadas."""
    con.execute("UPDATE empresas SET ativo = 0 WHERE cnpj = ?", (cnpj,))

    row = con.execute("SELECT ult_nsu FROM controle_nsu WHERE cnpj = ?", (cnpj,)).fetchone()
    return row[0] if row else NSU_ZERO


def salvar_ult_nsu(con: sqlite3.Connection, cnpj: str, nsu: str,
                   max_nsu: str | None = None) -> None:
    con.execute(
        """INSERT INTO controle_nsu (cnpj, ult_nsu, max_nsu, atualizado_em) VALUES (?, ?, ?, ?)
           ON CONFLICT(cnpj) DO UPDATE SET
               ult_nsu       = excluded.ult_nsu,
               max_nsu       = COALESCE(excluded.max_nsu, controle_nsu.max_nsu),
               atualizado_em = excluded.atualizado_em""",
        (cnpj, nsu.zfill(15), max_nsu.zfill(15) if max_nsu else None, _agora()),

def bloquear_empresa(con: sqlite3.Connection, cnpj: str, minutos: int = 60) -> datetime:
    """Chamar quando a SEFAZ retornar cStat 656 (consumo indevido)."""
    ate = datetime.now(FUSO) + timedelta(minutes=minutos)
    con.execute(
        """INSERT INTO controle_nsu (cnpj, ult_nsu, bloqueado_ate) VALUES (?, ?, ?)
           ON CONFLICT(cnpj) DO UPDATE SET bloqueado_ate = excluded.bloqueado_ate""",
        (cnpj, NSU_ZERO, ate.strftime(_FMT)),
    )
    return ate


def bloqueada_ate(con: sqlite3.Connection, cnpj: str) -> datetime | None:

    if not row or not row[0] or row[0] <= _agora():
        return None
    return datetime.strptime(row[0], _FMT).replace(tzinfo=FUSO)

def salvar_completo(con: sqlite3.Connection, r: Resumo, xml: str, pasta: Path,

               tipo         = 'completo',
               caminho_xml  = excluded.caminho_xml,

               situacao     = excluded.situacao,
               empresa_cnpj = COALESCE(nfe.empresa_cnpj, excluded.empresa_cnpj)""",
        (r.chave, empresa_cnpj, r.emitente_cnpj, r.emitente_nome, r.data_emissao,
         str(r.valor), r.situacao, r.protocolo, str(caminho)),

# ---------------------------------------------------------------- consulta da tela
def _condicoes(fornecedor: str, inicio: str | None, fim: str | None,
               empresa_cnpj: str | None = None):
    condicoes, parametros = [], []
    if empresa_cnpj:
        condicoes.append("empresa_cnpj = ?")
        parametros.append(empresa_cnpj)
    if fornecedor.strip():
        condicoes.append("emitente_nome LIKE ?")
        parametros.append(f"%{fornecedor.strip()}%")
    if inicio:
        condicoes.append("data_emissao >= ?")
        parametros.append(f"{inicio} 00:00:00")
    if fim:
        condicoes.append("data_emissao <= ?")
        parametros.append(f"{fim} 23:59:59")
    where = f"WHERE {' AND '.join(condicoes)}" if condicoes else ""
    return where, parametros


def buscar_pagina(con: sqlite3.Connection, fornecedor: str, inicio: str | None,
                  fim: str | None, limite: int, deslocamento: int,
                  empresa_cnpj: str | None = None) -> tuple[list[dict], dict]:
    """Retorna as notas da página e as métricas dos filtros ativos."""
    where, parametros = _condicoes(fornecedor, inicio, fim, empresa_cnpj)
    cur = _cursor(con)
    notas = cur.execute(
        f"""SELECT chave, empresa_cnpj, emitente_nome AS emitente, valor, data_emissao,
                   manifestacao, caminho_xml, caminho_xml IS NOT NULL AS xml_baixado
            FROM nfe {where}
            ORDER BY data_emissao DESC LIMIT ? OFFSET ?""",
        [*parametros, limite, deslocamento],
    ).fetchall()
    resumo = cur.execute(
        f"""SELECT COUNT(*) AS quantidade,
                   COALESCE(SUM(CAST(valor AS REAL)), 0) AS valor_total,
                   COALESCE(SUM(CASE WHEN caminho_xml IS NOT NULL THEN 1 ELSE 0 END), 0) AS baixados
            FROM nfe {where}""",
        parametros,
    ).fetchone()
    return [dict(n) for n in notas], dict(resumo)

import re
import subprocess
import tempfile
from pathlib import Path

from nfe_monitor.infrastructure.sefaz.parametros import Config


class ErroTransporte(RuntimeError):
    """Falha ao chamar a SEFAZ ou resposta HTTP inesperada."""


_LINHA_STATUS = re.compile(r"^HTTP/?\S*\s+(\d{3})\b")

# Existe só no Windows. Em outros sistemas, vale 0 (sem efeito).
_SEM_JANELA = getattr(subprocess, "CREATE_NO_WINDOW", 0)


def _envelope_distribuicao(cnpj: str, bloco_consulta: str, cfg: Config) -> str:
    return f"""<?xml version="1.0" encoding="utf-8"?>
<soap12:Envelope xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
                 xmlns:xsd="http://www.w3.org/2001/XMLSchema"
                 xmlns:soap12="http://www.w3.org/2003/05/soap-envelope">
  <soap12:Body>
    <nfeDistDFeInteresse xmlns="http://www.portalfiscal.inf.br/nfe/wsdl/NFeDistribuicaoDFe">
      <nfeDadosMsg>
        <distDFeInt xmlns="http://www.portalfiscal.inf.br/nfe" versao="1.01">
          <tpAmb>{cfg.tp_amb}</tpAmb>
          <cUFAutor>{cfg.uf_autor}</cUFAutor>
          <CNPJ>{cnpj}</CNPJ>
          {bloco_consulta}
        </distDFeInt>
      </nfeDadosMsg>
    </nfeDistDFeInteresse>
  </soap12:Body>
</soap12:Envelope>"""


def montar_envelope(cnpj: str, ult_nsu: str, cfg: Config) -> str:
    bloco = f"""<distNSU>
            <ultNSU>{ult_nsu}</ultNSU>
          </distNSU>"""
    return _envelope_distribuicao(cnpj, bloco, cfg)


def montar_envelope_chave(cnpj: str, chave: str, cfg: Config) -> str:
    bloco = f"""<consChNFe>
            <chNFe>{chave}</chNFe>
          </consChNFe>"""
    return _envelope_distribuicao(cnpj, bloco, cfg)


def _decodificar(bruto: bytes) -> str:
    """Converte bytes em texto, tratando BOM UTF-8 e UTF-16 (o que o PowerShell grava)."""
    if bruto.startswith((b"\xff\xfe", b"\xfe\xff")):
        texto = bruto.decode("utf-16")
    else:
        texto = bruto.decode("utf-8-sig", errors="replace")
    return texto.replace("\x00", "").lstrip("\ufeff")


def _executar_powershell(entrada: Path, saida: Path, cfg: Config) -> None:
    try:
        r = subprocess.run(
            [
                "powershell", "-NoProfile", "-ExecutionPolicy", "Bypass",
                "-File", str(cfg.script_ps),
                "-Thumbprint", cfg.thumbprint,
                "-Url", cfg.url,
                "-ArquivoEntrada", str(entrada),
                "-ArquivoSaida", str(saida),
                "-SoapAction", cfg.soap_action,
            ],
            capture_output=True,
            timeout=cfg.timeout_s,
            creationflags=_SEM_JANELA,
        )
    except subprocess.TimeoutExpired as e:
        raise ErroTransporte(f"Sem resposta da SEFAZ após {cfg.timeout_s}s") from e

    if r.returncode != 0:
        erro = _decodificar(r.stderr) + _decodificar(r.stdout)
        raise ErroTransporte(f"PowerShell falhou (código {r.returncode}):\n{erro.strip()}")

    if not saida.exists():
        raise ErroTransporte("O script PowerShell terminou sem gerar o arquivo de saída")


def _validar_resposta(texto: str) -> str:
    """Confere a linha de status e devolve somente o corpo XML."""
    texto = texto.lstrip()
    linha, _, corpo = texto.partition("\n")
    corpo = corpo.strip()

    m = _LINHA_STATUS.match(linha.strip())
    if not m:
        raise ErroTransporte(
            "Linha de status não reconhecida.\n"
            f"Início do arquivo: {texto[:200]!r}"
        )
    if m.group(1) != "200":
        raise ErroTransporte(f"HTTP {m.group(1)} da SEFAZ:\n{corpo[:500]}")
    if not corpo.startswith("<"):
        raise ErroTransporte(f"Corpo da resposta não é XML:\n{corpo[:300]}")
    return corpo


def enviar_soap(envelope: str, cfg: Config) -> str:
    """Envia um envelope SOAP já montado (e assinado, se for o caso) e devolve o corpo XML."""
    with tempfile.TemporaryDirectory() as pasta:
        entrada = Path(pasta) / "entrada.xml"
        saida = Path(pasta) / "saida.txt"

        entrada.write_text(envelope, encoding="utf-8")
        _executar_powershell(entrada, saida, cfg)
        texto = _decodificar(saida.read_bytes())

    return _validar_resposta(texto)


def consultar(cnpj: str, ult_nsu: str, cfg: Config) -> str:
    """Distribuição a partir do NSU (lotes)."""
    return enviar_soap(montar_envelope(cnpj, ult_nsu, cfg), cfg)


def consultar_chave(cnpj: str, chave: str, cfg: Config) -> str:
    """Distribuição pela chave (consChNFe). Devolve o XML completo se a ciência já foi registrada."""
    return enviar_soap(montar_envelope_chave(cnpj, chave, cfg), cfg)

import base64

import xml.etree.ElementTree as ET

__all__ = [
    "NS", "CSTAT_ESPERA", "CSTAT_SEM_DOCS", "CSTAT_COM_DOCS", "FUSO",
    "Documento", "Lote", "Resumo",
    "normalizar_data", "ler_lote", "ler_resumo", "ler_completo",
]

NS = {"nfe": "http://www.portalfiscal.inf.br/nfe"}

CSTAT_ESPERA = "656"      # consumo indevido: aguardar antes de nova consulta
CSTAT_SEM_DOCS = "137"    # nenhum documento localizado
CSTAT_COM_DOCS = "138"    # documentos localizados


def normalizar_data(texto: str | None) -> str:
    """Converte a data para 'AAAA-MM-DD HH:MM:SS' no horário de Brasília (ordenável como texto)."""
    if not texto:
        return ""
    try:
        dt = datetime.fromisoformat(texto)
    except ValueError:
        return texto
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=FUSO)
    return dt.astimezone(FUSO).strftime("%Y-%m-%d %H:%M:%S")


@dataclass(frozen=True)
class Documento:
    nsu: str
    schema: str
    xml: str

    @property
    def tipo(self) -> str:
        if self.schema.startswith("resNFe"):
            return "resumo"
        if self.schema.startswith("procNFe"):
            return "completo"
        return "outro"


@dataclass(frozen=True)
class Lote:
    cstat: str
    motivo: str
    ult_nsu: str | None
    max_nsu: str | None
    docs: list[Documento] = field(default_factory=list)


def _encontrar_ret(raiz: ET.Element) -> ET.Element | None:
    """Localiza retDistDFeInt na raiz ou em qualquer nível abaixo dela."""
    alvo = f"{{{NS['nfe']}}}retDistDFeInt"
    return next((e for e in raiz.iter() if e.tag == alvo), None)


def ler_lote(corpo: str, exigir_nsu: bool = True) -> Lote:
    ret = _encontrar_ret(ET.fromstring(corpo))
    if ret is None:
        raise ValueError("Resposta sem retDistDFeInt")

    cstat = ret.findtext("nfe:cStat", namespaces=NS)
    motivo = ret.findtext("nfe:xMotivo", namespaces=NS)
    ult_nsu = ret.findtext("nfe:ultNSU", namespaces=NS)
    max_nsu = ret.findtext("nfe:maxNSU", namespaces=NS)

    if not cstat:
        raise ValueError("Resposta da SEFAZ sem cStat")
    if exigir_nsu and cstat in (CSTAT_SEM_DOCS, CSTAT_COM_DOCS) and not (ult_nsu and max_nsu):
        raise ValueError(f"Resposta cStat {cstat} sem ultNSU ou maxNSU")

    docs = []
    for d in ret.findall("nfe:loteDistDFeInt/nfe:docZip", NS):
        conteudo = gzip.decompress(base64.b64decode(d.text or "")).decode("utf-8")
        docs.append(Documento(nsu=d.get("NSU", ""), schema=d.get("schema", ""), xml=conteudo))

    return Lote(cstat=cstat, motivo=motivo or "", ult_nsu=ult_nsu, max_nsu=max_nsu, docs=docs)


def ler_resumo(xml: str) -> Resumo:
    """resNFe: dados mínimos. Situação é cSitNFe (1 autorizada, 2 cancelada, 3 denegada)."""
    r = ET.fromstring(xml)
    return Resumo(
        chave=r.findtext("nfe:chNFe", namespaces=NS),
        emitente_cnpj=r.findtext("nfe:CNPJ", namespaces=NS) or "",
        emitente_nome=r.findtext("nfe:xNome", namespaces=NS) or "",
        data_emissao=normalizar_data(r.findtext("nfe:dhEmi", namespaces=NS)),
        valor=Decimal(r.findtext("nfe:vNF", namespaces=NS)),
        situacao=r.findtext("nfe:cSitNFe", namespaces=NS) or "",
        protocolo=r.findtext("nfe:nProt", namespaces=NS),
    )


def ler_completo(xml: str) -> Resumo:
    """procNFe: XML completo. Situação é o cStat do protocolo (100 autorizada, 101 cancelada)."""
    raiz = ET.fromstring(xml)
    inf = raiz.find(".//nfe:infNFe", NS)
    if inf is None:
        raise ValueError("procNFe sem infNFe")

    emit = inf.find("nfe:emit", NS)
    prot = raiz.find(".//nfe:protNFe/nfe:infProt", NS)

    return Resumo(
        chave=inf.get("Id", "")[3:],  # remove o prefixo "NFe"
        emitente_cnpj=(emit.findtext("nfe:CNPJ", namespaces=NS)
                       or emit.findtext("nfe:CPF", namespaces=NS) or ""),
        emitente_nome=emit.findtext("nfe:xNome", namespaces=NS) or "",
        data_emissao=normalizar_data(
            inf.findtext("nfe:ide/nfe:dhEmi", namespaces=NS)
            or inf.findtext("nfe:ide/nfe:dEmi", namespaces=NS)
        ),
        valor=Decimal(inf.findtext("nfe:total/nfe:ICMSTot/nfe:vNF", namespaces=NS)),
        situacao=(prot.findtext("nfe:cStat", namespaces=NS) or "") if prot is not None else "",
        protocolo=prot.findtext("nfe:nProt", namespaces=NS) if prot is not None else None,
    )

def _assinar(xml: str, thumbprint: str) -> str:

def _local(tag: str) -> str:

def _resultado_evento(corpo: str) -> tuple[str, str]:

    raiz = ET.fromstring(corpo)
    for el in raiz.iter():
        if _local(el.tag) != "infEvento":
            continue
        valores = {_local(f.tag): (f.text or "") for f in el}
        return valores.get("cStat", ""), valores.get("xMotivo", "")
    raise ErroEvento("Resposta do serviço de eventos sem infEvento")

from PySide6.QtCore import QAbstractListModel, QByteArray, QModelIndex, Qt

from nfe_monitor.ui.formatters import formatar_brl, formatar_data
from nfe_monitor.domain.nota import numero_da_nota, situacao_da

_TEXTO_SITUACAO = {"baixado": "XML baixado", "ciencia": "Ciência registrada", "pendente": "Pendente"}


def _iniciais(nome: str) -> str:
    partes = [p for p in nome.split() if p[:1].isalnum()]
    return "".join(p[0] for p in partes[:2]).upper() or "?"


class NotasModel(QAbstractListModel):
    _NOMES = ["emitente", "iniciais", "numero", "data", "valor",
              "situacao", "situacaoTexto", "chave", "baixado"]
    ROLES = {Qt.UserRole + i + 1: n for i, n in enumerate(_NOMES)}

    def __init__(self, parent=None) -> None:
        super().__init__(parent)
        self._notas: list[dict] = []

    def rowCount(self, parent=QModelIndex()) -> int:
        return 0 if parent.isValid() else len(self._notas)

    def roleNames(self) -> dict[int, QByteArray]:
        return {r: QByteArray(n.encode()) for r, n in self.ROLES.items()}

    def data(self, index: QModelIndex, role: int = Qt.DisplayRole):
        if not index.isValid():
            return None
        n = self._notas[index.row()]
        match self.ROLES.get(role):
            case "emitente": return n["emitente"] or "Emitente não informado"
            case "iniciais": return _iniciais(n["emitente"] or "")
            case "numero": return numero_da_nota(n["chave"])
            case "data": return formatar_data(n["data_emissao"])
            case "valor": return formatar_brl(n["valor"])
            case "situacao": return situacao_da(n)
            case "situacaoTexto": return _TEXTO_SITUACAO[situacao_da(n)]
            case "chave": return n["chave"]
            case "baixado": return bool(n["xml_baixado"])
        return None

    def carregar(self, notas: list[dict]) -> None:
        self.beginResetModel()
        self._notas = notas
        self.endResetModel()

    def nota(self, linha: int) -> dict | None:
        return self._notas[linha] if 0 <= linha < len(self._notas) else None

    function corAvatar(nome) {
        let h = 0
        for (let i = 0; i < nome.length; i++) h = (h * 31 + nome.charCodeAt(i)) >>> 0
        return paletaAvatar[h % paletaAvatar.length]
    }
    function paraIso(texto) {
        const d = texto.replace(/\D/g, "")
        return d.length === 8 ? d.slice(4) + "-" + d.slice(2, 4) + "-" + d.slice(0, 2) : ""
    }
    function mascaraData(campo) {
        const d = campo.text.replace(/\D/g, "").slice(0, 8)
        let f = d
        if (d.length > 4) f = d.slice(0, 2) + "/" + d.slice(2, 4) + "/" + d.slice(4)
        else if (d.length > 2) f = d.slice(0, 2) + "/" + d.slice(2)
        if (f !== campo.text) campo.text = f
    }
    function filtrar() {
        viewModel.aplicarFiltros(buscaInput.text, paraIso(deInput.text), paraIso(ateInput.text))
    }

from typing import Any

from PySide6.QtCore import Property, QObject, QSettings, Qt, Signal, Slot
from PySide6.QtGui import QGuiApplication

from nfe_monitor.ui.themes.catalog import SYSTEM_ID, THEMES, resolve_theme

_CHAVE = "aparencia/tema"


class ThemeManager(QObject):
    themeChanged = Signal()

    def __init__(self, app: QGuiApplication, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._settings = QSettings()
        self._hints = app.styleHints()
        valor = str(self._settings.value(_CHAVE, SYSTEM_ID))
        self._selecionado = valor if valor in THEMES else SYSTEM_ID
        self._hints.colorSchemeChanged.connect(self._ao_mudar_sistema)
        self._dados: dict[str, Any] = {}
        self._aplicar()

    def _sistema_escuro(self) -> bool:
        return self._hints.colorScheme() == Qt.ColorScheme.Dark

    def _aplicar(self) -> None:
        self._dados = resolve_theme(self._selecionado, self._sistema_escuro()).to_qml()
        self.themeChanged.emit()

    def _ao_mudar_sistema(self, *_: object) -> None:
        if self._selecionado == SYSTEM_ID:
            self._aplicar()

    @Property("QVariantMap", notify=themeChanged)
    def t(self) -> dict[str, Any]:
        return self._dados

    @Property(str, notify=themeChanged)
    def selecionado(self) -> str:
        return self._selecionado

    @Property("QVariantList", notify=themeChanged)
    def opcoes(self) -> list[dict[str, str]]:
        modo = "escuro" if self._sistema_escuro() else "claro"
        return [{"id": SYSTEM_ID, "nome": f"Seguir o sistema ({modo})"}] + [
            {"id": t.id, "nome": t.nome} for t in THEMES.values()
        ]

    @Slot(str)
    def selecionar(self, theme_id: str) -> None:
        if theme_id != SYSTEM_ID and theme_id not in THEMES:
            return
        self._selecionado = theme_id
        self._settings.setValue(_CHAVE, theme_id)
        self._aplicar()

    @Slot()
    def alternar(self) -> None:
        """Atalho da barra lateral: percorre os temas em ciclo."""
        ids = [SYSTEM_ID, *THEMES.keys()]
        self.selecionar(ids[(ids.index(self._selecionado) + 1) % len(ids)])

    if not c or len(c) != 14:
        return c or "sem CNPJ"
    return f"{c[:2]}.{c[2:5]}.{c[5:8]}/{c[8:12]}-{c[12:]}"

class ConfigViewModel(QObject):
    alterado = Signal()
    salvo = Signal()
    erro = Signal(str)
    cadastrado = Signal(str)

    def __init__(self, parent: QObject | None = None) -> None:
        super().__init__(parent)
        self._certs: list[dict] = []
        self._empresas: list[dict] = []
        self._thumb = ""
        self._pasta = ""
        self._ocupado = False
        self._tarefas: dict[QThread, _Tarefa] = {}

    # ---------- Propriedades ----------
    @Property("QVariantList", notify=alterado)
    def certificados(self) -> list[dict]: return self._certs

    @Property("QVariantList", notify=alterado)
    def empresas(self) -> list[dict]: return self._empresas

    @Property(str, notify=alterado)
    def thumbprint(self) -> str: return self._thumb

    @Property(str, notify=alterado)
    def pasta(self) -> str: return self._pasta

    @Property(bool, notify=alterado)
    def ocupado(self) -> bool: return self._ocupado

    # ---------- Carga ----------
    @Slot()
    def carregar(self) -> None:
        prefs = preferencias.carregar()
        self._thumb, self._pasta = prefs.thumbprint.upper(), prefs.pasta_download
        try:
            lista = certificados.listar()
        except RuntimeError as e:
            self.erro.emit(str(e))
            lista = []
        self._certs = [self._cert_dict(c) for c in lista]
        self._carregar_empresas()
        self.alterado.emit()

    def _cert_dict(self, c: certificados.Certificado) -> dict:
        marcas = []
        if c.thumbprint == self._thumb:
            marcas.append("ATUAL")
        if c.vencido:
            marcas.append("VENCIDO")
        elif c.dias_restantes <= certificados.AVISO_DIAS:
            marcas.append("VENCE LOGO")
        if not c.tem_chave:
            marcas.append("SEM CHAVE")
        extra = f"  [{', '.join(marcas)}]" if marcas else ""
        return {
            "id": c.thumbprint,
            "nome": f"{c.titular} · {_fmt_cnpj(c.cnpj)}{extra}",
            "cnpj": _fmt_cnpj(c.cnpj),
            "validade": f"{c.validade:%d/%m/%Y}",
            "situacao": c.situacao,
            "alerta": c.vencido or c.dias_restantes <= certificados.AVISO_DIAS or not c.tem_chave,
        }

    def _carregar_empresas(self) -> None:
        try:
            con = conectar(Config().banco)
            try:
                lista = repo.listar_empresas(con, somente_ativas=False)
            finally:
                con.close()
        except Exception as e:  # noqa: BLE001 - mostra na tela em vez de derrubar
            self.erro.emit(f"Não foi possível ler as empresas: {e}")
            lista = []
        self._empresas = [
            {
                "cnpj": _fmt_cnpj(e.cnpj),
                "nome": e.razao_social or "(sem nome)",
                "uf": e.uf or "--",
                "completa": bool(e.completa),
                "ativo": bool(e.ativo),
            }
            for e in lista
        ]

    # ---------- Cadastro ----------
    @Slot(str, str)
    def cadastrar(self, thumbprint: str, uf: str) -> None:
        if self._ocupado:
            return
        if not thumbprint:
            self.erro.emit("Selecione um certificado.")
            return
        uf = uf.strip().upper() or None
        self._ocupado = True
        self.alterado.emit()

        thread = QThread(self)
        tarefa = _Tarefa(lambda: empresas.cadastrar_por_certificado(
            thumbprint, uf=uf, definir_padrao=False))
        tarefa.moveToThread(thread)
        thread.started.connect(tarefa.run)
        tarefa.concluiu.connect(self._cadastro_ok)
        tarefa.falhou.connect(self._cadastro_erro)
        tarefa.concluiu.connect(thread.quit)
        tarefa.falhou.connect(thread.quit)
        thread.finished.connect(tarefa.deleteLater)
        thread.finished.connect(thread.deleteLater)
        thread.finished.connect(lambda: self._tarefas.pop(thread, None))
        self._tarefas[thread] = tarefa
        thread.start()

    def _cadastro_ok(self, res) -> None:
        e = res.empresa
        linhas = [f"✓ {'Cadastrada' if res.nova else 'Atualizada'}: "
                  f"{e.razao_social or '(sem nome)'} ({e.uf or 'sem UF'})"]
        linhas += [f"⚠ {a}" for a in res.avisos]
        self._ocupado = False
        self._carregar_empresas()
        self.alterado.emit()
        self.cadastrado.emit("\n".join(linhas))

    def _cadastro_erro(self, mensagem: str) -> None:
        self._ocupado = False
        self.alterado.emit()
        self.erro.emit(mensagem)

    # ---------- Preferências ----------
    @Slot(str, result=str)
    def caminhoDe(self, url: str) -> str:
        return str(Path(QUrl(url).toLocalFile()))

    @Slot(str, str, result=bool)
    def salvar(self, thumbprint: str, pasta: str) -> bool:
        if not thumbprint:
            self.erro.emit("Selecione um certificado.")
            return False
        prefs = preferencias.carregar()
        prefs.thumbprint, prefs.pasta_download = thumbprint, pasta
        preferencias.salvar(prefs)
        self._thumb, self._pasta = thumbprint.upper(), pasta
        self.alterado.emit()
        self.salvo.emit()
        return True

    @Property(int, notify=empresasChanged)

        for i, e in enumerate(self._empresas, start=1):
            if e.cnpj == self._empresa_filtro:
                return i
        return 0

    @Property(bool, notify=empresasChanged)
    def multiEmpresa(self) -> bool: return len(self._empresas) > 1

    @Property(int, notify=alterado)
    def pagina(self) -> int: return self._pagina + 1

    @Property(int, notify=alterado)
    def totalPaginas(self) -> int: return self._total_paginas

    @Property(int, notify=alterado)
    def totalNotas(self) -> int: return self._total

    @Property(str, notify=alterado)
    def valorTotal(self) -> str: return self._valor

    @Property(int, notify=alterado)
    def baixados(self) -> int: return self._baixados

    @Property(bool, notify=alterado)
    def ocupado(self) -> bool: return self._ocupado

    @Property(str, notify=alterado)
    def status(self) -> str: return self._status

    @Property(str, notify=alterado)
    def certTexto(self) -> str: return self._cert_texto

    @Property(bool, notify=alterado)
    def certAlerta(self) -> bool: return self._cert_alerta

    @Property(bool, notify=alterado)
    def filtrado(self) -> bool: return bool(self._fornecedor or self._inicio or self._fim)

    @Property(str, notify=relogioChanged)
    def proxima(self) -> str: return self._proxima

    @Property(str, notify=relogioChanged)
    def ultima(self) -> str: return self._ultima

    @Property(bool, notify=relogioChanged)
    def syncLiberado(self) -> bool: return self._liberado

    # ---------- Internos ----------
    def _set_status(self, texto: str, ocupado: bool | None = None) -> None:
        self._status = texto
        if ocupado is not None:
            self._ocupado = ocupado
        self.alterado.emit()
        self._tick()

    def _tick(self) -> None:
        restante = int(limite.segundos_restantes())
        ultima = limite.ultima_sincronizacao()
        self._ultima = f"{ultima:%d/%m/%Y às %H:%M}" if ultima else "Nunca sincronizado"
        if restante > 0:
            h, r = divmod(restante, 3600)
            m, s = divmod(r, 60)
            self._proxima = f"{h:02d}:{m:02d}:{s:02d}"
        else:
            self._proxima = ""
        self._liberado = restante == 0 and not self._ocupado
        self.relogioChanged.emit()

    def _carregar_empresas(self) -> None:
        try:
            con = conectar(Config().banco)
            try:
                self._empresas = repo.listar_empresas(con)
            finally:
                con.close()
        except Exception:  # noqa: BLE001 - sem banco, mantém o CNPJ do paths
            self._empresas = []

        # Empresa salva foi desativada/removida, ou só restou uma: volta para "todas"
        cnpjs = {e.cnpj for e in self._empresas}
        if self._empresa_filtro and (self._empresa_filtro not in cnpjs or len(cnpjs) <= 1):
            self._empresa_filtro = ""
            self._salvar_filtro()
        self.empresasChanged.emit()

    def _salvar_filtro(self) -> None:
        try:
            p = preferencias.carregar()
            p.empresa_filtro = self._empresa_filtro
            preferencias.salvar(p)
        except OSError:
            pass  # não salvar a preferência não impede o uso

    def _executar(self, funcao, ok, falha) -> None:
        thread = QThread(self)
        tarefa = Tarefa(funcao)
        tarefa.moveToThread(thread)
        thread.started.connect(tarefa.run)
        tarefa.concluiu.connect(ok)
        tarefa.falhou.connect(falha)
        tarefa.concluiu.connect(thread.quit)
        tarefa.falhou.connect(thread.quit)
        thread.finished.connect(tarefa.deleteLater)
        thread.finished.connect(thread.deleteLater)
        thread.finished.connect(lambda: self._tarefas.pop(thread, None))
        self._tarefas[thread] = tarefa
        thread.start()

    def _validar_cert(self) -> bool:
        try:
            certificados.validar_para_uso()
            return True
        except RuntimeError as e:
            self.erro.emit("Certificado", str(e))
            return False

    @staticmethod
    def _pasta_salva() -> Path | None:
        salva = preferencias.carregar().pasta_download
        return Path(salva) if salva and Path(salva).is_dir() else None

    # ---------- Consulta / filtros ----------
    def _consultar(self):
        return consultar_notas(self._fornecedor, self._inicio, self._fim,
                               self._pagina, self._empresa_filtro or None)

    @Slot()
    def recarregar(self) -> None:
        notas, resumo = self._consultar()
        self._total = int(resumo["quantidade"] or 0)
        self._total_paginas = max(1, -(-self._total // ITENS_POR_PAGINA))
        if self._pagina >= self._total_paginas:
            self._pagina = self._total_paginas - 1
            notas, resumo = self._consultar()
        self._valor = formatar_brl(resumo["valor_total"] or 0)
        self._baixados = int(resumo["baixados"] or 0)
        self._model.carregar(notas)
        self.alterado.emit()

    @Slot(str)
    def selecionarEmpresa(self, cnpj: str) -> None:
        cnpj = "".join(c for c in (cnpj or "") if c.isdigit())
        if cnpj == self._empresa_filtro or self._ocupado:
            self.empresasChanged.emit()  # devolve o seletor ao valor atual
            return
        self._empresa_filtro = cnpj
        self._salvar_filtro()
        self._pagina = 0
        self.empresasChanged.emit()
        self.recarregar()

    @Slot(str, str, str)
    def aplicarFiltros(self, fornecedor: str, inicio_iso: str, fim_iso: str) -> None:
        self._fornecedor = fornecedor.strip()
        self._inicio = inicio_iso or None
        self._fim = fim_iso or None
        self._pagina = 0
        self.recarregar()

    @Slot()
    def limparFiltros(self) -> None:
        self.aplicarFiltros("", "", "")

    @Slot(int)
    def irPagina(self, delta: int) -> None:
        nova = self._pagina + delta
        if 0 <= nova < self._total_paginas and not self._ocupado:
            self._pagina = nova
            self.recarregar()

    # ---------- Certificado ----------
    @Slot()
    def atualizarCertificado(self) -> None:
        try:
            cert = certificados.selecionado()
        except (RuntimeError, OSError, ValueError, KeyError):
            cert = None
        if cert is None:
            self._cert_texto, self._cert_alerta = "Certificado não configurado", True
        else:
            self._cert_texto = f"{cert.titular.split(':')[0]} · {cert.situacao} até {cert.validade:%d/%m/%Y}"
            self._cert_alerta = cert.dias_restantes <= certificados.AVISO_DIAS
        self._carregar_empresas()  # chamado ao salvar as Configurações
        self.recarregar()
        self.alterado.emit()

    # ---------- Sincronização ----------
    @Slot()
    def sincronizar(self) -> None:
        if self._ocupado or limite.segundos_restantes() > 0 or not self._validar_cert():
            return
        limite.registrar_disparo()
        self._set_status("Sincronizando com a SEFAZ…", True)
        self._executar(lambda: sincronizar_todas(Config()), self._sync_ok, self._sync_erro)

    def _sync_ok(self, resultados) -> None:
        if not resultados:
            self._set_status("Nenhuma empresa cadastrada", False)
            self.erro.emit("Sincronização", "Cadastre uma empresa em ⚙ Configurações.")
            return

        novas = sum(r.novas for r in resultados)
        esperas = [r for r in resultados if r.status == "espera"]
        erros = [r for r in resultados if r.status == "erro"]

        partes = [f"{novas} nova(s)"]
        if esperas:
            partes.append(f"{len(esperas)} em espera")
        if erros:
            partes.append(f"{len(erros)} com erro")
        resumo = " · ".join(partes)

        self._carregar_empresas()
        self._set_status(f"Concluído: {resumo}", False)
        self.recarregar()
        self.notificacao.emit(f"✓ {resumo}")

        if erros:
            self.erro.emit("Sincronização",
                           "\n".join(f"{_fmt_cnpj(r.cnpj)}: {r.detalhe}" for r in erros))

    def _sync_erro(self, mensagem: str) -> None:
        self._set_status("Falha na sincronização", False)
        self.recarregar()
        self.erro.emit("Sincronização", mensagem)

    # ---------- XML ----------
    @Slot(int)
    def acaoXml(self, linha: int) -> None:
        if self._ocupado or self._model.nota(linha) is None:
            return
        pasta = self._pasta_salva()
        if pasta is None:
            self.solicitarPasta.emit(linha)
        else:
            self._prosseguir(linha, pasta)

    @Slot(int, str)
    def continuarComPasta(self, linha: int, url: str) -> None:
        self._prosseguir(linha, Path(QUrl(url).toLocalFile()))

    def _prosseguir(self, linha: int, pasta: Path) -> None:
        nota = self._model.nota(linha)
        if nota is None:
            return
        if nota["xml_baixado"]:
            origem = Path(nota["caminho_xml"] or "")
            if not origem.is_file():
                self.erro.emit("Salvar XML", f"O arquivo não foi encontrado em:\n{origem}")
                return
            try:
                destino = caminho_livre(pasta, f"{nota['chave']}.xml")
                shutil.copy2(origem, destino)
            except OSError as e:
                self.erro.emit("Salvar XML", f"Não foi possível salvar o arquivo:\n{e}")
                return
            self._set_status(f"XML salvo em {destino}")
            self.notificacao.emit("✓ XML salvo")
            return

        if not self._validar_cert():
            return
        if nota["manifestacao"] != "ciencia":
            self.solicitarCiencia.emit(linha, str(pasta))
            return
        self._baixar(nota["chave"], pasta)

    @Slot(int, str)
    def confirmarCiencia(self, linha: int, pasta: str) -> None:
        nota = self._model.nota(linha)
        if nota is not None:
            self._baixar(nota["chave"], Path(pasta))

    def _baixar(self, chave: str, pasta: Path) -> None:
        self._set_status("Baixando XML…", True)
        self._executar(lambda: baixar_e_copiar(chave, pasta), self._download_ok, self._download_erro)

    def _download_ok(self, destino: Path) -> None:
        self._set_status(f"XML salvo em {destino}", False)
        self.recarregar()
        self.notificacao.emit("✓ XML baixado e salvo")

    def _download_erro(self, mensagem: str) -> None:
        self._set_status("Falha ao baixar o XML", False)
        self.recarregar()
        self.erro.emit("Baixar XML", mensagem)

    # ---------- Janela ----------
    @Slot(result=bool)
    def podeFechar(self) -> bool:
        return not self._ocupado

# 📄 NF-e Monitor

Aplicativo desktop para **monitorar, consultar e baixar as NF-e emitidas contra os CNPJs da sua empresa**. Ele usa o serviço de **Distribuição DF-e da SEFAZ** e acessa a SEFAZ com o certificado digital A1 instalado no Windows.

Ele foi feito para escritórios e empresas que precisam acompanhar as notas de entrada de **um ou mais CNPJs** num só lugar.

---

## ✨ Funcionalidades

- 🔄 **Sincronização com a SEFAZ** a partir do último NSU, respeitando o intervalo obrigatório de 1 hora
- 🏢 **Várias empresas**: sincroniza todos os CNPJs ativos, e a tela filtra as notas por empresa
- 🔐 **Certificado digital do Windows**: cada empresa pode usar o seu certificado ou o certificado padrão
- 📥 **Download do XML completo** com manifestação de **Ciência da Operação**
- 🔍 **Filtros** por fornecedor, período de emissão e empresa
- 📊 **Painel** com quantidade de notas, valor total e progresso dos XMLs baixados
- 🎨 **Temas** claro e escuro
- 🧾 **Registro de operações**: cada sincronização fica gravada no banco, com início, fim, status e detalhe

---

## 🧱 Arquitetura

O projeto é organizado em camadas, no estilo *Clean Architecture* com *MVVM* na interface:

```
nfe-monitor/
├── main.py                      # Ponto de entrada (QML + view models)
├── requirements.txt
└── nfe_monitor/
    ├── domain/                  # Regras e entidades (Resumo, FUSO)
    ├── application/             # Casos de uso
    │   ├── sincronizar.py       # Distribuição DF-e (uma ou todas as empresas)
    │   ├── listar_notas.py      # Consulta paginada + métricas
    │   ├── baixar_xml.py        # Manifestação + download do XML
    │   └── limite.py            # Controle do intervalo entre sincronizações
    ├── infrastructure/
    │   ├── database/
    │   │   ├── conexao.py       # Conexão SQLite + códigos de UF
    │   │   └── notas_repository.py
    │   ├── sefaz/
    │   │   ├── cliente.py       # Chamadas SOAP
    │   │   ├── distribuicao.py  # Leitura dos lotes (cStat, NSU, docs)
    │   │   └── parametros.py    # Config (banco, thumbprint, UF)
    │   ├── certificados.py      # Certificados do repositório do Windows
    │   ├── preferencias.py      # config.json do usuário
    │   └── paths.py             # Pastas de dados e recursos (compatível com PyInstaller)
    └── ui/
        ├── viewmodels/
        │   ├── main_viewmodel.py    # Notas, filtros, sincronização
        │   └── config_viewmodel.py  # Certificados, empresas, preferências
        ├── themes/
        │   └── theme_manager.py     # Tema claro/escuro (exposto como "Theme")
        ├── models/notas_model.py
        ├── formatters.py
        └── qml/
            └── Main.qml             # Interface (Qt Quick, estilo Basic)
```

### Como a interface é montada

O `main.py` cria os três objetos e os expõe ao QML como propriedades de contexto:

| Nome no QML | Classe Python | Função |
|---|---|---|
| `Theme` | `ThemeManager` | Cores e tema ativo |
| `viewModel` | `MainViewModel` | Lista de notas, filtros, sincronização e download |
| `config` | `ConfigViewModel` | Tela de configurações |

Quando você salva as configurações, o sinal `config.salvo` chama `viewModel.atualizarCertificado`, então o certificado novo passa a valer na hora, sem reiniciar o programa.

### Tecnologias

| Camada | Bibliotecas |
|---|---|
| Interface | **PySide6 6.12** (Qt Quick / QML, estilo *Basic*) |
| Comunicação com a SEFAZ | **httpx** (HTTP), **lxml** (XML), **xsdata** (classes geradas a partir dos XSD) |
| Certificado digital | **cryptography** |
| Banco de dados | **SQLite** (já vem com o Python) |
| Qualidade | **pytest**, **ruff**, **mypy** |

## ⚙️ Configuração inicial

Na primeira execução, clique em **⚙ Configurações** na barra lateral.

### 1. Certificado digital
- Escolha o certificado na lista. Use **↻** para recarregar a lista se você acabou de instalar um certificado.
- Abaixo da lista aparecem o **CNPJ**, a **validade** e a **situação** do certificado. Ele fica em alerta quando está perto de vencer.

### 2. Cadastrar empresas
1. Selecione o certificado da empresa.
2. Preencha a **UF** ou deixe em branco para buscar na **BrasilAPI**.
3. Clique em **＋ Cadastrar empresa do certificado selecionado**.

Cada empresa aparece com um status:

| Status | Significado |
|---|---|
| 🟢 **Pronta** | Tem UF e certificado, já pode sincronizar |
| 🟡 **Incompleta** | Falta a UF ou o certificado |
| ⚪ **Inativa** | Não é mais monitorada, mas as notas dela continuam no banco |

### 3. Pasta dos XMLs
- **Escolher**: define uma pasta padrão para onde os XMLs são copiados.
- **Limpar**: faz o programa perguntar a pasta a cada download.

> Independentemente dessa escolha, o programa sempre guarda uma cópia do XML em `%APPDATA%\NFeMonitor\xml\<CNPJ>\`.

### 4. Aparência
Escolha o tema. Também dá para alternar pelo botão ☀/☾ da barra lateral.

Clique em **Salvar** para aplicar. A mudança vale na hora, sem reiniciar.

## 📏 Regras da SEFAZ que o sistema respeita

| Situação | O que o sistema faz |
|---|---|
| Fila zerada (`cStat 137` ou `ultNSU == maxNSU`) | Bloqueia novas consultas daquela empresa por **60 min** |
| Consumo indevido (`cStat 656`) | Bloqueia por **60 min** e registra a ocorrência |
| Lote processado | Grava as notas e o NSU **na mesma transação**, então nada se perde se der falha no meio |
| NSU não avançou | Interrompe a consulta para evitar um laço infinito |

## 📦 Gerar executável (PyInstaller)

O `paths.py` já encontra os arquivos de recurso (QML) tanto no código-fonte quanto dentro do executável, usando `sys._MEIPASS`.

```powershell
pip install pyinstaller
pyinstaller main.py --name "NFe Monitor" --windowed --noconfirm ^
  --add-data "nfe_monitor/ui/qml;nfe_monitor/ui/qml"
```

| Opção | Para quê |
|---|---|
| `--windowed` | Abre sem a janela preta do console |

## 🛠️ Solução de problemas

| Problema | Solução |
|---|---|
| A janela não abre (código de saída `1`) | Erro no `Main.qml`. Veja a mensagem no terminal |
| `ModuleNotFoundError: nfe_monitor` | Rode o comando a partir da raiz do projeto, com o `.venv` ativado |
| "Certificado não configurado" | Selecione e salve um certificado em ⚙ Configurações |
| Certificado não aparece na lista | Instale o `.pfx` em *Usuário Atual → Pessoal* e clique em ↻ |
| Empresa **Incompleta** | Cadastre de novo informando a UF manualmente |
| `0 nova(s) · 1 em espera` | Comportamento normal: a SEFAZ exige esperar 1 h |
| Notas antigas somem ao filtrar por empresa | São notas sem `empresa_cnpj`. Corrija com o SQL abaixo |
| "Há uma operação em andamento" ao fechar | Aguarde a sincronização ou o download terminar |
| Erro ao instalar o `PySide6` | Confira se o Python é **64 bits** e se a versão é 3.10 ou superior |
| A tarefa agendada não encontra as notas | Configure a tarefa para rodar com o mesmo usuário do Windows que usa o aplicativo |

"""Teste da camada de certificados do Windows.

Somente leitura: lista os certificados do repositório Pessoal do usuário
atual, extrai o CNPJ, a validade e confirma se há chave privada.
Não faz chamadas de rede e não usa senha.
"""
import ctypes
import re
import sys
from ctypes import wintypes
from datetime import datetime, timezone

from cryptography import x509
from cryptography.hazmat.primitives import hashes
from cryptography.x509.oid import ExtensionOID, NameOID, ObjectIdentifier

if sys.platform != "win32":
    raise SystemExit("Este teste só roda no Windows.")

crypt32 = ctypes.WinDLL("crypt32", use_last_error=True)

CERT_SYSTEM_STORE_CURRENT_USER = 0x00010000
CERT_STORE_READONLY_FLAG = 0x00008000
CERT_STORE_PROV_SYSTEM_W = 10
CERT_KEY_PROV_INFO_PROP_ID = 2
OID_CNPJ = ObjectIdentifier("2.16.76.1.3.3")


class CERT_CONTEXT(ctypes.Structure):
    _fields_ = [
        ("dwCertEncodingType", wintypes.DWORD),
        ("pbCertEncoded", ctypes.c_void_p),
        ("cbCertEncoded", wintypes.DWORD),
        ("pCertInfo", ctypes.c_void_p),
        ("hCertStore", ctypes.c_void_p),
    ]


crypt32.CertOpenStore.argtypes = [
    ctypes.c_void_p, wintypes.DWORD, ctypes.c_void_p, wintypes.DWORD, ctypes.c_wchar_p,
]
crypt32.CertOpenStore.restype = ctypes.c_void_p

crypt32.CertEnumCertificatesInStore.argtypes = [ctypes.c_void_p, ctypes.c_void_p]
crypt32.CertEnumCertificatesInStore.restype = ctypes.c_void_p

crypt32.CertGetCertificateContextProperty.argtypes = [
    ctypes.c_void_p, wintypes.DWORD, ctypes.c_void_p, ctypes.POINTER(wintypes.DWORD),
]
crypt32.CertGetCertificateContextProperty.restype = wintypes.BOOL

crypt32.CertCloseStore.argtypes = [ctypes.c_void_p, wintypes.DWORD]
crypt32.CertCloseStore.restype = wintypes.BOOL


def tem_chave_privada(contexto: int) -> bool:
    tamanho = wintypes.DWORD(0)
    return bool(crypt32.CertGetCertificateContextProperty(
        contexto, CERT_KEY_PROV_INFO_PROP_ID, None, ctypes.byref(tamanho),
    ))


def extrair_cnpj(certificado: x509.Certificate) -> str | None:
    try:
        san = certificado.extensions.get_extension_for_oid(
            ExtensionOID.SUBJECT_ALTERNATIVE_NAME
        ).value
    except x509.ExtensionNotFound:
        return None
    for nome in san:
        if isinstance(nome, x509.OtherName) and nome.type_id == OID_CNPJ:
            achados = re.findall(rb"\d{14}", nome.value)
            if achados:
                return achados[0].decode()
    return None


def listar_certificados() -> list[tuple[x509.Certificate, bool]]:
    loja = crypt32.CertOpenStore(
        CERT_STORE_PROV_SYSTEM_W, 0, None,
        CERT_SYSTEM_STORE_CURRENT_USER | CERT_STORE_READONLY_FLAG, "MY",
    )
    if not loja:
        raise OSError(ctypes.get_last_error(), "Falha ao abrir o repositório MY.")

    encontrados = []
    try:
        contexto = crypt32.CertEnumCertificatesInStore(loja, None)
        while contexto:
            ctx = ctypes.cast(contexto, ctypes.POINTER(CERT_CONTEXT)).contents
            der = ctypes.string_at(ctx.pbCertEncoded, ctx.cbCertEncoded)
            cert = x509.load_der_x509_certificate(der)
            encontrados.append((cert, tem_chave_privada(contexto)))
            contexto = crypt32.CertEnumCertificatesInStore(loja, contexto)
    finally:
        crypt32.CertCloseStore(loja, 0)
    return encontrados


def nome_comum(nome: x509.Name) -> str:
    atributos = nome.get_attributes_for_oid(NameOID.COMMON_NAME)
    return atributos[0].value if atributos else nome.rfc4514_string()


def mascarar_cnpj(cnpj: str | None) -> str:
    if not cnpj:
        return "não encontrado"
    return f"{cnpj[:2]}.***.***/****-{cnpj[-2:]}"


def main() -> None:
    agora = datetime.now(timezone.utc)
    certificados = listar_certificados()
    print(f"Certificados no repositório Pessoal do usuário: {len(certificados)}\n")

    for i, (cert, tem_chave) in enumerate(certificados, start=1):
        dias = (cert.not_valid_after_utc - agora).days
        print(f"[{i}] {nome_comum(cert.subject)}")

        print(f"    Emissor:         {nome_comum(cert.issuer)}")

        print(f"    Thumbprint:      {cert.fingerprint(hashes.SHA1()).hex()[:8]}...")

"""Consulta de status ao NFeDistribuicaoDFe em homologação (tpAmb=2).

Localiza o certificado no repositório Pessoal do usuário pelo CNPJ e envia

def erro_win(operacao: str) -> OSError:
    codigo = ctypes.get_last_error()
    return OSError(codigo, f"{operacao} falhou (código Windows {codigo})")


def tem_chave_privada(contexto: int) -> bool:
    tamanho = DWORD(0)
    return bool(crypt32.CertGetCertificateContextProperty(
        contexto, CERT_KEY_PROV_INFO_PROP_ID, None, ctypes.byref(tamanho),
    ))


def extrair_cnpj(certificado: x509.Certificate) -> str | None:
    try:
        san = certificado.extensions.get_extension_for_oid(
            ExtensionOID.SUBJECT_ALTERNATIVE_NAME
        ).value
    except x509.ExtensionNotFound:
        return None
    for nome in san:
        if isinstance(nome, x509.OtherName) and nome.type_id == OID_CNPJ:
            achados = re.findall(rb"\d{14}", nome.value)
            if achados:
                return achados[0].decode()
    return None


def localizar_certificado(cnpj: str) -> int:
    """Retorna um CERT_CONTEXT duplicado (liberar com CertFreeCertificateContext)."""
    loja = crypt32.CertOpenStore(
        CERT_STORE_PROV_SYSTEM_W, 0, None,
        CERT_SYSTEM_STORE_CURRENT_USER | CERT_STORE_READONLY_FLAG, "MY",
    )
    if not loja:
        raise erro_win("CertOpenStore")
    try:
        ctx = crypt32.CertEnumCertificatesInStore(loja, None)
        while ctx:
            info = ctypes.cast(ctx, ctypes.POINTER(CERT_CONTEXT)).contents
            der = ctypes.string_at(info.pbCertEncoded, info.cbCertEncoded)
            certificado = x509.load_der_x509_certificate(der)
            if extrair_cnpj(certificado) == cnpj and tem_chave_privada(ctx):
                escolhido = crypt32.CertDuplicateCertificateContext(ctx)
                crypt32.CertFreeCertificateContext(ctx)
                return escolhido
            ctx = crypt32.CertEnumCertificatesInStore(loja, ctx)
        raise LookupError(
            f"Nenhum certificado com chave privada para o CNPJ {cnpj[:2]}***********"
        )
    finally:
        crypt32.CertCloseStore(loja, 0)


def montar_envelope(cnpj: str, uf: str) -> bytes:
    xml = f"""<?xml version="1.0" encoding="utf-8"?>
<soap12:Envelope xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
                 xmlns:xsd="http://www.w3.org/2001/XMLSchema"
                 xmlns:soap12="http://www.w3.org/2003/05/soap-envelope">
  <soap12:Body>
    <nfeDistDFeInteresse xmlns="http://www.portalfiscal.inf.br/nfe/wsdl/NFeDistribuicaoDFe">
      <nfeDadosMsg>
        <distDFeInt xmlns="http://www.portalfiscal.inf.br/nfe" versao="1.01">
          <tpAmb>2</tpAmb>
          <cUFAutor>{uf}</cUFAutor>
          <CNPJ>{cnpj}</CNPJ>
          <distNSU>
            <ultNSU>000000000000000</ultNSU>
          </distNSU>
        </distDFeInt>
      </nfeDadosMsg>
    </nfeDistDFeInteresse>
  </soap12:Body>
</soap12:Envelope>"""
    return xml.encode("utf-8")


def enviar(corpo: bytes, cert_ctx: int) -> tuple[int, bytes]:
    sessao = winhttp.WinHttpOpen(
        "nfe_monitor/teste", WINHTTP_ACCESS_TYPE_DEFAULT_PROXY, None, None, 0,
    )
    if not sessao:
        raise erro_win("WinHttpOpen")
    try:
        protocolos = DWORD(WINHTTP_FLAG_SECURE_PROTOCOL_TLS1_2)
        winhttp.WinHttpSetOption(
            sessao, WINHTTP_OPTION_SECURE_PROTOCOLS,
            ctypes.addressof(protocolos), ctypes.sizeof(protocolos),
        )
        winhttp.WinHttpSetTimeouts(sessao, 15000, 15000, 30000, 60000)

        conexao = winhttp.WinHttpConnect(sessao, HOST, 443, 0)
        if not conexao:
            raise erro_win("WinHttpConnect")
        try:
            req = winhttp.WinHttpOpenRequest(
                conexao, "POST", CAMINHO, None, None, None, WINHTTP_FLAG_SECURE,
            )
            if not req:
                raise erro_win("WinHttpOpenRequest")
            try:
                if not winhttp.WinHttpSetOption(
                    req, WINHTTP_OPTION_CLIENT_CERT_CONTEXT,
                    cert_ctx, 0,
                ):
                    raise erro_win("WinHttpSetOption(certificado de cliente)")

                cabecalhos = (
                    "Content-Type: application/soap+xml; charset=utf-8; "
                    f'action="{ACTION}"'
                )
                if not winhttp.WinHttpSendRequest(
                    req, cabecalhos, 0xFFFFFFFF, corpo, len(corpo), len(corpo), None,
                ):
                    raise erro_win("WinHttpSendRequest")
                if not winhttp.WinHttpReceiveResponse(req, None):
                    raise erro_win("WinHttpReceiveResponse")

                status = DWORD(0)
                tamanho = DWORD(ctypes.sizeof(status))
                winhttp.WinHttpQueryHeaders(
                    req, WINHTTP_QUERY_STATUS_CODE | WINHTTP_QUERY_FLAG_NUMBER, None,
                    ctypes.addressof(status), ctypes.byref(tamanho), None,
                )

                resposta = bytearray()
                disponivel = DWORD(0)
                while True:
                    if not winhttp.WinHttpQueryDataAvailable(req, ctypes.byref(disponivel)):
                        raise erro_win("WinHttpQueryDataAvailable")
                    if disponivel.value == 0:
                        break
                    buf = ctypes.create_string_buffer(disponivel.value)
                    lidos = DWORD(0)
                    if not winhttp.WinHttpReadData(
                        req, buf, disponivel.value, ctypes.byref(lidos),
                    ):
                        raise erro_win("WinHttpReadData")
                    resposta += buf.raw[: lidos.value]
                return status.value, bytes(resposta)
            finally:
                winhttp.WinHttpCloseHandle(req)
        finally:
            winhttp.WinHttpCloseHandle(conexao)
    finally:
        winhttp.WinHttpCloseHandle(sessao)


def resumir(corpo: bytes) -> None:
    texto = corpo.decode("utf-8", errors="replace")
    cstat = re.search(r"<cStat>(\d+)</cStat>", texto)
    motivo = re.search(r"<xMotivo>(.*?)</xMotivo>", texto, re.S)
    ult = re.search(r"<ultNSU>(\d+)</ultNSU>", texto)
    print(f"  cStat:    {cstat.group(1) if cstat else 'não encontrado'}")
    print(f"  xMotivo:  {motivo.group(1).strip() if motivo else 'não encontrado'}")
    print(f"  ultNSU:   {ult.group(1) if ult else 'não encontrado'}")
    print("\nTrecho inicial da resposta:")
    print(texto[:800])


def main() -> None:
    parser = argparse.ArgumentParser(description="Status do NFeDistribuicaoDFe em homologação.")
    parser.add_argument("--cnpj", required=True, help="CNPJ da empresa (14 dígitos)")
    parser.add_argument("--uf", default="29", help="Código IBGE da UF (BA = 29)")
    args = parser.parse_args()

    cnpj = re.sub(r"\D", "", args.cnpj)
    if len(cnpj) != 14:
        raise SystemExit("CNPJ deve ter 14 dígitos.")

    print("Localizando certificado no repositório...")
    cert_ctx = localizar_certificado(cnpj)
    try:
        print(f"Enviando consulta para {HOST} (tpAmb=2, distNSU=0)...")
        status, corpo = enviar(montar_envelope(cnpj, args.uf), cert_ctx)
    finally:
        crypt32.CertFreeCertificateContext(cert_ctx)

    print(f"\nHTTP {status}")
    resumir(corpo)


if __name__ == "__main__":
    main()

"""
Gera um único arquivo com a estrutura e o conteúdo do projeto,
pronto para ser enviado a uma IA.

Uso:
    python tools/gerar_snapshot.py
    python tools/gerar_snapshot.py --saida snapshot.txt --max-kb 200
"""

from __future__ import annotations

import argparse
from pathlib import Path

RAIZ = Path(__file__).resolve().parent.parent

PASTAS_IGNORADAS = {
    ".venv", "venv", ".git", "__pycache__", ".pytest_cache",
    ".mypy_cache", ".ruff_cache", ".idea", ".vscode",
    "build", "dist", "htmlcov", ".tox", "node_modules",
}

EXTENSOES_INCLUIDAS = {
    ".py", ".toml", ".cfg", ".ini", ".md", ".txt",
    ".yaml", ".yml", ".json", ".spec",
}

ARQUIVOS_IGNORADOS = {"snapshot_projeto.txt", ".coverage"}


def deve_ignorar(caminho: Path) -> bool:
    partes = set(caminho.relative_to(RAIZ).parts)
    if partes & PASTAS_IGNORADAS:
        return True
    if any(p.endswith(".egg-info") for p in partes):
        return True
    return caminho.name in ARQUIVOS_IGNORADOS


def gerar_arvore(arquivos: list[Path]) -> str:
    linhas = [f"{RAIZ.name}/"]
    for arquivo in arquivos:
        relativo = arquivo.relative_to(RAIZ)
        nivel = len(relativo.parts) - 1
        linhas.append(f"{'    ' * nivel}{relativo.name}")
    return "\n".join(linhas)


def coletar_arquivos() -> list[Path]:
    return sorted(
        p for p in RAIZ.rglob("*")
        if p.is_file()
        and not deve_ignorar(p)
        and p.suffix.lower() in EXTENSOES_INCLUIDAS
    )


def ler_conteudo(arquivo: Path, max_bytes: int) -> str:
    if arquivo.stat().st_size > max_bytes:
        return f"[arquivo omitido: maior que {max_bytes // 1024} KB]"
    try:
        return arquivo.read_text(encoding="utf-8")
    except UnicodeDecodeError:
        return arquivo.read_text(encoding="latin-1")


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--saida", default="snapshot_projeto.txt")
    parser.add_argument("--max-kb", type=int, default=200)
    args = parser.parse_args()

    arquivos = coletar_arquivos()
    saida = RAIZ / args.saida
    max_bytes = args.max_kb * 1024

    with saida.open("w", encoding="utf-8") as f:
        f.write("# ESTRUTURA DO PROJETO\n\n")
        f.write(gerar_arvore(arquivos))
        f.write("\n\n# CONTEÚDO DOS ARQUIVOS\n")

        for arquivo in arquivos:
            relativo = arquivo.relative_to(RAIZ).as_posix()
            f.write(f"\n\n{'=' * 80}\n")
            f.write(f"ARQUIVO: {relativo}\n")
            f.write(f"{'=' * 80}\n")
            f.write(ler_conteudo(arquivo, max_bytes))

    tamanho_kb = saida.stat().st_size / 1024
    print(f"✔ {len(arquivos)} arquivos -> {saida.name} ({tamanho_kb:.1f} KB)")


if __name__ == "__main__":
    main()
